# Hadoop安装环境简易手册

## 前情提要

相关路径注意改动与自己的环境适配，本手册仅为示范

## 运行环境

- 操作系统：Linux
- 软件环境：jdk-8，Hadoop-3.2.1
- 资源需求：2核vCPU，4G内存，20G硬盘

## 安装教程（任务1）

```bash
# 1. 操作系统安装，提供一个选择，安装windows系统下的wsl，参考链接如下:
https://zhuanlan.zhihu.com/p/337104547

# 2. 安装SSH
sudo apt-get install ssh
sudo apt-get install rsync

# 3. 安装 Java 8
# 更新软件，并安装
sudo apt update
sudo apt install openjdk-8-jdk -y

# 验证安装
java -version

# 4. 安装Hadoop - 3.2.1
# 回到你的用户主目录
cd ~
# 建立安装目录并进入 
mkdir -p ~/hadoop_installs
cd ~/hadoop_installs

# 下载 Hadoop 3.2.1 压缩包 (从 Apache 官方归档库下载)(这一步如果因为网络问题失败，可以自行从浏览器下载，然后移动到wsl所在目录中，wsl目录在文件资源管理器中找到Linux图标，通常和C、D盘并列)
wget https://archive.apache.org/dist/hadoop/common/hadoop-3.2.1/hadoop-3.2.1.tar.gz

# 解压文件
tar -zxvf hadoop-3.2.1.tar.gz

# 5. 配置环境变量，可以使用任意工具，本手册使用vim
vim ~/.bashrc

# 然后移动光标到最末尾，按 i 进入编辑状态，粘贴如下信息：
export JAVA_HOME=/usr/lib/jvm/java-8-openjdk-amd64
export HADOOP_HOME=$HOME/hadoop_installs/hadoop-3.2.1
export PATH=$JAVA_HOME/bin:$HADOOP_HOME/bin:$HADOOP_HOME/sbin:$PATH
export CLASSPATH=$JAVA_HOME/lib:.

# 按 Esc 键，然后输出 :wq ， 按 enter 保存退出

# 让配置立即生效
source ~/.bashrc

# 验证是否配置成功
hadoop version

# 6.配置免密码SSH登录，Hadoop 伪分布模式需要你的系统能自己免密SSH登录自己，用来管理节点
# 生成 RSA 密钥，遇到提示一路狂按回车即可 (不需要输入密码)
ssh-keygen -t rsa -P '' -f ~/.ssh/id_rsa

# 将生成的密钥加入授权列表
cat ~/.ssh/id_rsa.pub >> ~/.ssh/authorized_keys

# 赋予授权文件正确的权限
chmod 0600 ~/.ssh/authorized_keys

# 测试登录
ssh localhost

# 可能遇到的问题，显示ssh: connect to host localhost port 22: Connection refused，这说明你的ssh服务没启动
# 启动ssh
sudo service ssh start # 然后再ssh localhost

# 7.修改Hadoop核心配置文件，依旧使用vim，具体保存操作等不再赘述
cd ~/hadoop_installs/hadoop-3.2.1/etc/hadoop
# 配置java路径
vim hadoop-env.sh
# 在文件末尾输入如下信息
export JAVA_HOME=/usr/lib/jvm/java-8-openjdk-amd64

# 配置core-site.xml
vim core-site.xml
# 将文件中<configuration>和</configuration>中的内容如下替换：
<configuration>
    <property>
        <name>fs.defaultFS</name>
        <value>hdfs://localhost:9000</value>
    </property>
    <property>
        <name>hadoop.tmp.dir</name>
        <value>/home/mt_linux/hadoop_installs/hadoop_tmp</value>
    </property>
</configuration>

# 配置hdfs-site.xml
vim hdfs-site.xml
# 将文件中<configuration>和</configuration>中的内容如下替换：
<configuration>
    <property>
        <name>dfs.replication</name>
        <value>1</value>
    </property>
</configuration>

# 配置mapred-site.xml
vim mapred-site.xml
# 将文件中<configuration>和</configuration>中的内容如下替换：
<configuration>
    <property>
        <name>mapreduce.framework.name</name>
        <value>yarn</value>
    </property>
    <property>
        <name>mapreduce.application.classpath</name>
        <value>/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/mapreduce/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/mapreduce/lib/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/common/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/common/lib/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/hdfs/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/hdfs/lib/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/yarn/*:/home/mt_linux/hadoop_installs/hadoop-3.2.1/share/hadoop/yarn/lib/*</value>
    </property>
    <property>
        <name>yarn.app.mapreduce.am.env</name>
        <value>HADOOP_MAPRED_HOME=/home/mt_linux/hadoop_installs/hadoop-3.2.1</value>
    </property>
    <property>
        <name>mapreduce.map.env</name>
        <value>HADOOP_MAPRED_HOME=/home/mt_linux/hadoop_installs/hadoop-3.2.1</value>
    </property>
    <property>
        <name>mapreduce.reduce.env</name>
        <value>HADOOP_MAPRED_HOME=/home/mt_linux/hadoop_installs/hadoop-3.2.1</value>
    </property>
</configuration>


# 配置yarn-site.xml
vim yarn-site.xml
<configuration>
    <property>
        <name>yarn.nodemanager.aux-services</name>
        <value>mapreduce_shuffle</value>
    </property>
    <property>
        <name>yarn.nodemanager.env-whitelist</name>
        <value>JAVA_HOME, HADOOP_COMMON_HOME, HADOOP_HDFS_HOME, HADOOP_CONF_DIR, CLASSPATH_PREPEND_DISTCACHE, HADOOP_YARN_HOME, HADOOP_MAPRED_HOME</value>
    </property>
</configuration>

# 8. 格式化 NameNode，切记，本条命令只能执行一次，多次执行格式化会导致混乱，因此非要第二次格式化时，得删除第一次格式化时创建的工作目录
cd ~/hadoop_installs/hadoop-3.2.1
bin/hdfs namenode -format # 最后提示 Storage directory ... has been successfully formatted. 或者 SHUTDOWN_MSG，就说明成功了

# 9 & 10 步: 启动Hadoop 集群，遇见warning可以查查ai，通常可以默认不管

# 启动 HDFS 进程 (NameNode, DataNode, SecondaryNameNode)
sbin/start-dfs.sh

# 启动 YARN 进程 (ResourceManager, NodeManager)
sbin/start-yarn.sh

# 查看所有启动的Java进程
jps

# 查看HDFS存储管理界面（从浏览器登录，需要事先启动集群）
http://localhost:9870

# 查看 YARN 资源调度界面
http://localhost:8088

```



## 任务2

```bash
# 在 HDFS 中创建你当前用户的专属目录和输入文件夹 test-in(注意，路径要适配你自己的路径)
bin/hdfs dfs -mkdir -p /user/mt_linux/test-in

# 在你本地的 WSL 系统中创建一个临时目录存放测试数据
mkdir -p ~/test_data

# 创建两个包含单词的文本文件
echo "hello hadoop hello world" > ~/test_data/file1.txt
echo "goodbye hadoop" > ~/test_data/file2.txt

# 将本地的这两个 txt 文件上传到 HDFS 的 test-in 目录中
bin/hdfs dfs -put ~/test_data/*.txt /user/mt_linux/test-in/

# 可以看看 HDFS 里是不是真的有这两个文件了：
bin/hdfs dfs -ls /user/mt_linux/test-in/

# 运行MapReduce任务
bin/hadoop jar share/hadoop/mapreduce/hadoop-mapreduce-examples-3.2.1.jar wordcount /user/mt_linux/test-in /user/mt_linux/test-out

# 查看计算结果
bin/hdfs dfs -cat /user/mt_linux/test-out/part-r-00000

# 实验做完之后安全关闭Hadoop 集群
# 停止 YARN 资源调度服务
sbin/stop-yarn.sh

# 停止 HDFS 存储服务
sbin/stop-dfs.sh

```
